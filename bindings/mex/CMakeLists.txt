# ── nxr-compute-mex — MATLAB MEX wrapper around the nxr-compute library ──────
#
# Single mexFunction with command dispatch. Built only when MATLAB
# is detected at configure time; absent MATLAB the rest of the
# native workspace (nxr-compute, addon, cli) builds unaffected.
#
# Usage from MATLAB:
#   ops = nxr_compute('assembleManifoldOperators', V, F);
#   eig = nxr_compute('solveEigenmodes', ops.cotanLaplacian, ops.mass, 300);

find_package(Matlab COMPONENTS MX_LIBRARY MAIN_PROGRAM)

if(NOT Matlab_FOUND)
    message(STATUS "nxr-compute-mex: MATLAB not found, skipping MEX target.")
    return()
endif()

message(STATUS "nxr-compute-mex: building against MATLAB at ${Matlab_ROOT_DIR}")

matlab_add_mex(
    NAME nxr_compute_mex                # CMake target name (file is nxr_compute.mexw64)
    SRC src/nxr_compute_mex.cpp
    LINK_TO nxr_compute
    OUTPUT_NAME nxr_compute
    R2018a                      # use the new mxArray API (interleaved complex etc.)
)

# matlab_add_mex doesn't always honour transitive compile features
# from LINK_TO, so set the standard explicitly.  nxr-compute is C++17 today
# (see nxr-compute/CMakeLists.txt for the Emscripten/Eigen reasoning).
target_compile_features(nxr_compute_mex PRIVATE cxx_std_17)

# libut.lib provides utIsInterruptPending() — the canonical Ctrl-C
# bridge inside MEX functions. Undocumented but stable across MATLAB
# releases for ~20 years (used by libigl-matlab, gptoolbox). Without
# it, long-running nxr-compute operations can't be interrupted from MATLAB.
# FindMatlab doesn't expose libut by default, so we resolve it manually.
if(WIN32)
    set(_matlab_libut_dir "${Matlab_ROOT_DIR}/extern/lib/win64/microsoft")
elseif(APPLE)
    # MATLAB R2023a+ ships native Apple Silicon under bin/maca64; older
    # x86_64 / Rosetta MATLAB lives under bin/maci64. Pick by build arch.
    if(CMAKE_OSX_ARCHITECTURES MATCHES "arm64" OR
       CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        set(_matlab_libut_dir "${Matlab_ROOT_DIR}/bin/maca64")
    else()
        set(_matlab_libut_dir "${Matlab_ROOT_DIR}/bin/maci64")
    endif()
else()
    set(_matlab_libut_dir "${Matlab_ROOT_DIR}/bin/glnxa64")
endif()
find_library(Matlab_UT_LIBRARY
    NAMES libut ut
    PATHS "${_matlab_libut_dir}"
    NO_DEFAULT_PATH
)
if(Matlab_UT_LIBRARY)
    message(STATUS "nxr-compute-mex: libut at ${Matlab_UT_LIBRARY}")
    # Plain signature — matlab_add_mex internally uses plain
    # target_link_libraries(nxr_compute_mex …) without keywords, so all later
    # links to the same target must be plain too.
    target_link_libraries(nxr_compute_mex "${Matlab_UT_LIBRARY}")
else()
    message(WARNING
        "nxr-compute-mex: libut not found in ${_matlab_libut_dir} — "
        "Ctrl-C will not interrupt nxr-compute solvers from MATLAB. "
        "Install path may differ on your platform.")
endif()

# macOS: drop FindMatlab's `cppMexFunction.map` from the link.
#
# CMake's FindMatlab unconditionally adds two `-Wl,-exported_symbols_list,…`
# flags on macOS when the MATLAB SDK supports the C++ MEX API (R2018a+):
#
#   • c_exportsmexfileversion.map  →  exports  _mexFunction, _mexfilerequiredapiversion
#                                     (legacy C-API entry point — what we use)
#   • cppMexFunction.map           →  ALSO requires the dylib export
#                                     _mexCreateMexFunction / _mexDestroyMexFunction /
#                                     _mexFunctionAdapter (the C++-class API entry
#                                     points generated by deriving from
#                                     matlab::mex::Function)
#
# Our source `nxr_compute_mex.cpp` is C-style (defines `void mexFunction(...)`),
# so the cpp map's required exports are never produced and the link fails.
# FindMatlab's classic ld64 workaround was to add `-Wl,-U,_mexCreate…` per
# symbol so they could remain undefined at link time, but Apple Silicon's
# ld-prime (Xcode 15+) treats them as initial-undefines that must still
# resolve and rejects the link.
#
# The cleanest fix is to strip cppMexFunction.map from this target's link
# line. Done via a post-`matlab_add_mex` LINK_FLAGS rewrite — we can't
# tell the FindMatlab macro not to add it. Scope: macOS-only; the
# Linux / Windows paths are untouched and still link as before.
if(APPLE)
    get_target_property(_mex_link_flags nxr_compute_mex LINK_FLAGS)
    if(_mex_link_flags)
        # Drop  -Wl,-exported_symbols_list,<…>cppMexFunction.map  in either form
        # FindMatlab might emit (with or without surrounding whitespace), and
        # drop the now-redundant per-symbol -U flags that pair with it.
        string(REGEX REPLACE
            "[ ]?-Wl,-exported_symbols_list,[^ ]*cppMexFunction\\.map" ""
            _mex_link_flags "${_mex_link_flags}")
        string(REGEX REPLACE
            "[ ]?-Wl,-U,_mex(CreateMexFunction|DestroyMexFunction|FunctionAdapter)" ""
            _mex_link_flags "${_mex_link_flags}")
        set_target_properties(nxr_compute_mex PROPERTIES LINK_FLAGS "${_mex_link_flags}")
    endif()
endif()

# Strip libMatlabEngine and libMatlabDataArray from the link line.
#
# FindMatlab's `matlab_add_mex(R2018a ...)` macro unconditionally links
# the C++ MEX API libraries (libMatlabEngine, libMatlabDataArray) when
# the SDK supports R2018a+. Our source `nxr_compute_mex.cpp` is a
# classic C-style mexFunction (uses `mex.h`, not `matlab::mex::Function`),
# so we never call into the C++ MEX runtime — but the link still
# records a runtime dependency.
#
# On macOS, the dependency carries the runner's build-time @rpath
# (e.g., the GitHub Actions MATLAB install dir). On a user's machine,
# that rpath is invalid and loading the .mexmaca64 fails with:
#   "Library not loaded: @rpath/libMatlabEngine.dylib"
# Linux and Windows have analogous failures (LD_LIBRARY_PATH / DLL
# search). The fix is to remove these libs from LINK_LIBRARIES on all
# platforms — they're entirely unneeded by a C-style mexFunction.
get_target_property(_mex_link_libs nxr_compute_mex LINK_LIBRARIES)
if(_mex_link_libs)
    # Match full paths, bare names, and CMake imported-target forms
    # (e.g. /…/libMatlabEngine.dylib, libMatlabEngine.so,
    #  MatlabEngine.lib, Matlab::MatlabEngine).
    list(FILTER _mex_link_libs EXCLUDE REGEX "MatlabEngine")
    list(FILTER _mex_link_libs EXCLUDE REGEX "MatlabDataArray")
    set_target_properties(nxr_compute_mex PROPERTIES LINK_LIBRARIES "${_mex_link_libs}")
endif()

# Also clear INTERFACE_LINK_LIBRARIES on the same target — some
# FindMatlab versions populate INTERFACE deps that get propagated.
get_target_property(_mex_iface_libs nxr_compute_mex INTERFACE_LINK_LIBRARIES)
if(_mex_iface_libs)
    list(FILTER _mex_iface_libs EXCLUDE REGEX "MatlabEngine")
    list(FILTER _mex_iface_libs EXCLUDE REGEX "MatlabDataArray")
    set_target_properties(nxr_compute_mex PROPERTIES INTERFACE_LINK_LIBRARIES "${_mex_iface_libs}")
endif()
